5 Week 5: Ocean Carbon Chemistry

In the first part of this tutorial, we use eigenanalysis to assess how dynamic air–sea carbon exchange shapes the pre-industrial climate equilibrium. We conduct two experiments: one with dynamic air–sea carbon exchange turned off and one with the mechanism turned on. Comparing the eigenanalysis results from these two experiments reveals how dynamic air–sea carbon exchange affects the climate equilibrium. In the second part of the tutorial, we simulate the effects of historical anthropogenic CO\(_2\) emissions on ocean carbon chemistry.

5.1 Eigenanalysis

5.1.1 Dynamic air-sea carbon exchange turned off

Let’s setup the model configuration.

rm(list = ls())

# Load the QUEST package:
library(quest)

# Set the path to the configuration files:
config.path <- "/global/teaching-project/sg5112000/shared_files/config"

# Load the configuration files:
config <- config_load(path = config.path)

# Set the flags to what we want
config$flags$run_AirSeaCarbon <- FALSE
config$flags$run_atmosphere <- TRUE
config$flags$run_ocean <- TRUE
config$flags$run_npzd <- FALSE
config$flags$run_land <- FALSE
config$flags$run_sic <- FALSE
config$flags$run_icesheets <- FALSE
config$flags$anthropogenic_emissions <- FALSE
config$flags$prescribed_CO2 <- FALSE
config$flags$prescribed_seaice <- FALSE
config$flags$diagnose_carbonates <- FALSE

Now we spin up the model for 10000 years.

times <- seq(-10000, 0, by = 1)
output <- core_run_esm(config = config, times = times)
saveRDS(output, "output_W05_config_01.rds")

# Create initialization file from this spinup
utils_write_new_init(
  input_file = "output_W05_config_01.rds",
  output_file = "config_init_W05_config_01.R",
  final_year = TRUE
)

Now we conduct the eigenanalysis for this model configuration.

# Get the spun up state variables
state_eq <- source("config_init_W05_config_01.R")$value

# Select state variables that will be perturbed
select_variables <- c("T_atm_L", "T_atm_H", "T_ocn_SL", "T_ocn_SH", "T_ocn_DL", "T_ocn_DH")

# Find the equilibria
equilibria <- eval_equilibria(config, state_eq, select_variables = select_variables, n_starts = 100)

# Conduct eigenanalysis
stability_results <- eval_eigenvalues(config,
  equilibria_dimensionless = equilibria$dimensionless,
  state_eq = state_eq,
  state_min_active = equilibria$state_min_active,
  state_max_active = equilibria$state_max_active,
  delta_time = 50
)
stability_results_AirSeaCarbon_off <- stability_results
saveRDS(stability_results_AirSeaCarbon_off, "W05_stability_results_AirSeaCarbon_off.rds")

5.1.3 Comparing the Jacobians

Let us now compare the results from both simulations

# Air-sea carbon exchange turned on:
ASC_on <- readRDS("W05_stability_results_AirSeaCarbon_on.rds")

# Air-sea carbon exchange turned off:
ASC_off <- readRDS("W05_stability_results_AirSeaCarbon_off.rds")

# Jacobian
delta_jacobian <- ASC_on$equilibrium_1$jacobian - ASC_off$equilibrium_1$jacobian
print(round(ASC_on$equilibrium_1$jacobian, 3))
print(round(ASC_off$equilibrium_1$jacobian, 3))
print(round(delta_jacobian, 3))

5.2 Anthropogenic Emissions

Let us now simulate the effects of historical anthropogenic carbon emissions on ocean carbon chemistry. To do this, we need to make two changes to the model configuration: enable anthropogenic emissions and activate the carbonate-system diagnostics. The following chunk runs the simulation from 1800 to 2024. It may take a few minutes to complete because the carbonate-system diagnostics add substantial computational overhead.

# Get the spun-up state
state_eq <- source("config_init_W05_config_02.R")$value

# Change model configuration
config$flags$anthropogenic_emissions <- TRUE
config$flags$diagnose_carbonates <- TRUE

# Run model and save output
times <- seq(1800, 2024, by = 1)
output <- core_run_esm(config = config, times = times)
saveRDS(output, "output_W05_config_03.rds")

Let us now plot the first set of variables.

output <- readRDS("output_W05_config_03.rds")
utils_plot(config, output, variables = c(
